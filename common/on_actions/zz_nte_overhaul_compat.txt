# Naval Treaty Expanded - Compatibility Hook for Naval Overhaul Mod (10-slot / Arms Race)
# Appends to on_actions list instead of overwriting naval_overhaul_on_ship_constructed's own effect block.

naval_overhaul_on_ship_constructed = {
	on_actions = {
		nte_overhaul_compat_pulse
	}
}

on_game_started = {
	on_actions = {
		nte_overhaul_startup_scan
	}
}

on_monthly_pulse_country = {
	on_actions = {
		nte_overhaul_monthly_scan
	}
}

nte_overhaul_compat_pulse = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = nte_naval_overhaul_detected } }
			set_global_variable = nte_naval_overhaul_detected
		}
		if = {
			limit = {
				has_global_variable = gnl_formed
				has_variable = gnl_participating_country
				NOT = { has_variable = nte_overhaul_flavor_received }
				OR = {
					has_technology_researched = dreadnought_tech
					AND = {
						exists = scope:ship
						scope:ship = {
							ship_group = ship_group:ship_group_capital_ships
							NOT = { is_ship_type = ship_type:ship_type_coastal_defense_ship }
							OR = {
								hit_points >= 8000
								power_projection >= 680
							}
						}
					}
				}
			}
			set_variable = nte_overhaul_flavor_received
			trigger_event = { id = nte_global.10 days = 3 }
		}
	}
}

nte_overhaul_startup_scan = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = nte_naval_overhaul_detected }
				# Overhaul sets this on game start; monthly scan covers load-order races.
				has_global_variable = naval_overhaul_active
			}
			set_global_variable = nte_naval_overhaul_detected
		}
	}
}

nte_overhaul_monthly_scan = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = nte_naval_overhaul_detected }
				has_global_variable = naval_overhaul_active
			}
			set_global_variable = nte_naval_overhaul_detected
		}
		# Soft-compat bridge: pause Overhaul's arms race while bound by Global or Bilateral Treaty (unless Escalator Clause invoked)
		if = {
			limit = {
				OR = {
					has_global_variable = nte_naval_overhaul_detected
					has_global_variable = naval_overhaul_active
				}
			}
			if = {
				limit = {
					OR = {
						AND = {
							has_variable = gnl_participating_country
							NOT = { has_variable = nte_escalator_clause_active }
						}
						has_variable = nte_bilateral_restricted_flag
					}
				}
				set_variable = overhaul_arms_race_paused_by_treaty
			}
			else_if = {
				limit = { has_variable = overhaul_arms_race_paused_by_treaty }
				remove_variable = overhaul_arms_race_paused_by_treaty
			}
		}
		if = {
			limit = {
				has_global_variable = nte_naval_overhaul_detected
				has_global_variable = gnl_formed
				has_variable = gnl_participating_country
				NOT = { has_variable = nte_overhaul_flavor_received }
				any_country = {
					is_country_alive = yes
					OR = {
						has_technology_researched = dreadnought_tech
						any_scope_ship = {
							ship_group = ship_group:ship_group_capital_ships
							NOT = { is_ship_type = ship_type:ship_type_coastal_defense_ship }
							OR = {
								hit_points >= 8000
								power_projection >= 680
							}
						}
					}
				}
			}
			set_variable = nte_overhaul_flavor_received
			trigger_event = { id = nte_global.10 days = 3 }
		}
	}
}
