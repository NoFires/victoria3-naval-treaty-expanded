# Naval Treaty Expanded - 1.15 Real Tonnage (Ship Era + Hit Points Complexity) & Voting Weights

# Single ship tonnage value evaluated in `ship` scope
# Capital ships count full tonnage (10-55 base + uncapped 10-slot mods); Coastal Defense / Pocket Battleships have a dedicated light-capital tier; Combat Cruisers are scaled ~35% with 10,000t Treaty Cruiser surcharge; Troop Transports, Destroyers/Submarines & Supply Ships count 0
nte_single_ship_effective_tonnage = {
	value = 0
	if = {
		limit = { is_ship_type = ship_type:ship_type_coastal_defense_ship }
		add = 14
		if = {
			limit = { power_projection > 130 }
			add = {
				value = power_projection
				subtract = 130
				divide = 30
				min = 0
				max = 10
			}
		}
		if = {
			limit = { hit_points > 1400 }
			add = {
				value = hit_points
				subtract = 1400
				divide = 300
				min = 0
				max = 8
			}
		}
	}
	else_if = {
		limit = { ship_group = ship_group:ship_group_capital_ships }
		if = {
			limit = { is_ship_type = ship_type:ship_type_monitor }
			add = 10
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_ship_of_the_line }
			add = 12
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_early_ironclad }
			add = 18
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_modern_ironclad }
			add = 24
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_pre_dreadnought }
			add = 30
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_dreadnought }
			add = 40
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_super_dreadnought }
			add = 55
		}
		else = {
			add = 20
		}
		# Firepower / propulsion / doctrine complexity via power_projection (calibrated for 10-slot upgrades)
		if = {
			limit = { power_projection > 150 }
			add = {
				value = power_projection
				subtract = 150
				divide = 22
				min = 0
				max = 55
			}
		}
		# Armor / hull durability premium via hit_points (calibrated for 10-slot upgrades)
		if = {
			limit = { hit_points > 2000 }
			add = {
				value = hit_points
				subtract = 2000
				divide = 220
				min = 0
				max = 50
			}
		}
	}
	else_if = {
		limit = {
			ship_group = ship_group:ship_group_cruisers
			NOR = {
				is_ship_type = ship_type:ship_type_early_troop_ship
				is_ship_type = ship_type:ship_type_troop_ship
			}
		}
		if = {
			limit = { is_ship_type = ship_type:ship_type_frigate }
			add = 3
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_iron_frigate }
			add = 5
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_protected_cruiser }
			add = 6
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_armored_cruiser }
			add = 8
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_light_cruiser }
			add = 9
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_seaplane_tender }
			add = 13
		}
		else_if = {
			limit = { is_ship_type = ship_type:ship_type_aircraft_carrier }
			add = 22
		}
		else = {
			add = 6
		}
		if = {
			limit = { power_projection > 120 }
			add = {
				value = power_projection
				subtract = 120
				divide = 25
				min = 0
				max = 22
			}
		}
		if = {
			limit = { hit_points > 1200 }
			add = {
				value = hit_points
				subtract = 1200
				divide = 220
				min = 0
				max = 18
			}
		}
		# 10,000t Treaty Cruiser qualitative surcharge: super-cruisers lose Jeune Ecole discount and pay capital-grade penalty
		if = {
			limit = {
				NOR = {
					is_ship_type = ship_type:ship_type_aircraft_carrier
					is_ship_type = ship_type:ship_type_seaplane_tender
				}
				OR = {
					hit_points >= 3400
					power_projection >= 380
				}
			}
			add = 8
		}
		else_if = {
			limit = {
				owner ?= { has_law_or_variant = law_type:law_jeune_ecole }
			}
			multiply = 0.72
		}
	}
}

# Total effective treaty tonnage of a country (evaluated in `country` scope)
nte_country_effective_tonnage = {
	value = 0
	every_scope_ship = {
		add = nte_single_ship_effective_tonnage
	}
	# Additive footprint reduction (capped at -28% max to prevent stacking exploits)
	multiply = nte_tonnage_reduction_factor
	round = yes
}

nte_tonnage_reduction_factor = {
	value = 1.0
	if = {
		limit = { has_modifier = nte_mothballed_fleet_modifier }
		subtract = 0.18
	}
	if = {
		limit = { has_variable = nte_scrapping_program_active }
		subtract = 0.12
	}
	if = {
		limit = { has_variable = nte_tonnage_falsification_active }
		subtract = 0.12
	}
	min = 0.62
}

# Era-scaled baseline tonnage bounds (evaluated in Chair / Rank 1 country scope)

		limit = { has_technology_researched = pre_dreadnought_tech }

		add = 100
	}
	if = {
		limit = { has_technology_researched = carrier_tech }

		limit = { has_technology_researched = dreadnought_tech }

	if = {
		limit = { has_technology_researched = carrier_tech }

# Mixed voting weight: Base seat vote (20) + 0.5 * Effective Tonnage (min 20, max 250)
nte_country_vote_weight = {
	value = 20
	if = {
		limit = { has_variable = nte_current_tonnage }
		add = {
			value = var:nte_current_tonnage
			multiply = 0.5
		}
	}
	else = {
		add = {
			value = nte_country_effective_tonnage
			multiply = 0.5
		}
	}
	if = {
		limit = { has_law_or_variant = law_type:law_diplomatic_navy }
		multiply = 1.15
	}
	min = 20
	max = 250
	round = yes
}

# Cached O(1) effective tonnage lookup for AI acceptance, treaty triggers, and non-member scans
nte_country_cached_effective_tonnage = {
	value = 0
	if = {
		limit = { has_variable = nte_cached_tonnage }
		value = var:nte_cached_tonnage
	}
	else_if = {
		limit = { has_variable = nte_current_tonnage }
		value = var:nte_current_tonnage
	}
	else = {
		value = nte_country_effective_tonnage
	}
}

# Effective sea power metric for treaty acceptance (combines combat displacement + screen/flotilla hull presence)
nte_diplomatic_naval_strength = {
	value = nte_country_cached_effective_tonnage
	add = {
		value = navy_size
		multiply = 3
	}
	min = 1
}

# Country legal tonnage cap under Global Naval Limitation Treaty
nte_prospective_tier_ratio = {
	value = 1.00
	if = {
		limit = { has_variable = nte_locked_tier_ratio }
		value = var:nte_locked_tier_ratio
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_global_variable = gnl_formed }
				NOT = { has_global_variable = gnl_tiers_initialized }
			}
		}
		if = {
			limit = { country_rank >= rank_value:great_power }
			value = 1.67
		}
		else = {
			value = 1.00
		}
	}
	else = {
		if = {
			limit = {
				country_rank >= rank_value:great_power
				OR = {
					NOT = { exists = global_var:gnl_rank1_country }
					NOT = { exists = global_var:gnl_rank2_country }
				}
			}
			value = 1.67
		}
		else_if = {
			limit = { NOT = { exists = global_var:gnl_rank3_country } }
			value = 1.00
		}
		else_if = {
			limit = {
				OR = {
					NOT = { exists = global_var:gnl_rank4_country }
					NOT = { exists = global_var:gnl_rank5_country }
				}
			}
			value = 0.58
		}
		else = {
			value = 0.45
		}
	}
}

nte_global_legal_limit = {
	value = 100
	if = {
		limit = {
			exists = global_var:gnl_tonnage_base_limit
			has_variable = nte_locked_tier_ratio
		}
		value = global_var:gnl_tonnage_base_limit
		multiply = var:nte_locked_tier_ratio
	}
	# Isolation sanction reduces allowed cap by 25%
	if = {
		limit = { has_modifier = nte_isolated_sanction_modifier }
		multiply = 0.75
	}
	# Escalator clause adds +20% emergency allowance
	if = {
		limit = { has_modifier = nte_escalator_clause_modifier }
		multiply = 1.20
	}
	min = 40
	round = yes
}

# Personal quota usage percentage (0 - 120+)
nte_personal_quota_usage_pct = {
	value = 0
	if = {
		limit = {
			has_variable = nte_current_tonnage
			has_variable = nte_current_legal_limit
			var:nte_current_legal_limit > 0
		}
		value = var:nte_current_tonnage
		divide = var:nte_current_legal_limit
		multiply = 100
		min = 0
		max = 120
	}
}

# Estimated share of total fixed expenses spent specifically on the NAVY (0.0 - 1.0)
# Weights each ship at 1.5x an army battalion to separate naval upkeep from land-heavy militaries
nte_estimated_naval_budget_share = {
	value = 0
	if = {
		limit = { navy_size > 0 }
		value = military_expenses_share
		multiply = {
			value = navy_size
			multiply = 1.5
			divide = {
				value = navy_size
				multiply = 1.5
				add = army_size
				min = 1
			}
		}
		min = 0
		max = 1
	}
}

# Unified fiscal distress factor (0.0 when healthy/high reserves; rises sharply with depleted reserves, deficit, and debt)
nte_fiscal_distress_multiplier = {
	value = 0

	# Credit/debt strain (non-linear: mild at low debt, severe near default)
	if = {
		limit = { scaled_debt > 0 }
		add = {
			value = scaled_debt
			multiply = scaled_debt
			multiply = 2.5
		}
		add = {
			value = scaled_debt
			multiply = 0.8
		}
	}

	# Ongoing fiscal deficit, cushioned by gold reserves (abundant reserves absorb temporary deficits)
	if = {
		limit = {
			fixed_income > 0
			net_fixed_income < 0
		}
		add = {
			value = net_fixed_income
			multiply = -1
			divide = {
				value = fixed_income
				min = 1
			}
			min = 0
			max = 0.50
			multiply = 2.0
			multiply = {
				value = 1.0
				subtract = {
					value = scaled_gold_reserves
					multiply = 1.5
					min = 0
					max = 1.0
				}
				min = 0
				max = 1.0
			}
		}
	}

	min = 0
	max = 2.5
}

# Effective naval fiscal pressure index P = estimated naval budget share * fiscal distress factor
nte_naval_fiscal_pressure_index = {
	value = nte_estimated_naval_budget_share
	multiply = nte_fiscal_distress_multiplier
}

# Acceptance score from unified naval fiscal pressure (0 when healthy/cushioned, 2..45 under genuine fiscal strain)
nte_naval_fiscal_acceptance_score = {
	value = 0
	if = {
		limit = {
			navy_size >= 5
			nte_naval_fiscal_pressure_index >= 0.012
		}
		value = nte_naval_fiscal_pressure_index
		multiply = 180
		min = 2
		max = 45
		round = yes
	}
}
