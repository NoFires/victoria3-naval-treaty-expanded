# Single Source of Truth for Global Treaty Stability (0..100)
nte_sync_global_stability_bars = {
	if = {
		limit = { NOT = { has_global_variable = gnl_treaty_stability } }
		set_global_variable = { name = gnl_treaty_stability value = 75 }
	}
	set_global_variable = {
		name = gnl_treaty_stability
		value = {
			value = global_var:gnl_treaty_stability
			min = 0
			max = 100
		}
	}
	every_country = {
		limit = { has_journal_entry = global_naval_limitation_union }
		je:global_naval_limitation_union ?= {
			set_bar_progress = {
				value = global_var:gnl_treaty_stability
				name = nte_treaty_stability_bar
			}
		}
	}
}
