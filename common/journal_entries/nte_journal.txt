	on_monthly_pulse = {
		effect = {
			# Also migrates members from older saves that only have the overview entry.
			nte_add_gnl_journal_entries = yes
			# Treaty climate sits on this member's journal entry, so withdrawing or dissolution removes it.
			scope:journal_entry = {
				if = {
					limit = {
						has_global_variable = gnl_treaty_stability
						global_var:gnl_treaty_stability >= 70
					}
					if = {
						limit = { NOT = { has_modifier = nte_treaty_concord_modifier } }
						add_modifier = nte_treaty_concord_modifier
					}
				}
				else_if = {
					limit = { has_modifier = nte_treaty_concord_modifier }
					remove_modifier = nte_treaty_concord_modifier
				}
				if = {
					limit = {
						has_global_variable = gnl_treaty_stability
						global_var:gnl_treaty_stability <= 30
					}
					if = {
						limit = { NOT = { has_modifier = nte_treaty_strain_modifier } }
						add_modifier = nte_treaty_strain_modifier
					}
				}
				else_if = {
					limit = { has_modifier = nte_treaty_strain_modifier }
					remove_modifier = nte_treaty_strain_modifier
				}
			}
			# Only the chair country drives the global monthly calculation to avoid N-times duplication in multiplayer
			if = {
				limit = {
					exists = global_var:global_gnl_chair_country
					global_var:global_gnl_chair_country = {
						has_variable = gnl_participating_country
					}
					this = global_var:global_gnl_chair_country
				}
				nte_global_treaty_monthly_pulse = yes
			}
			# Recovery fallback: if the chair was annexed or lost membership unexpectedly, highest-prestige survivor assumes chair
			else_if = {
				limit = {
					has_global_variable = gnl_formed
					has_variable = gnl_participating_country
					OR = {
						NOT = { exists = global_var:global_gnl_chair_country }
						NOT = {
							global_var:global_gnl_chair_country = {
								is_country_alive = yes
								has_variable = gnl_participating_country
							}
						}
					}
				}
				ordered_country = {
					limit = {
						is_country_alive = yes
						has_variable = gnl_participating_country
					}
					order_by = prestige
					position = 0
					set_global_variable = {
						name = global_gnl_chair_country
						value = this
					}
					# Only run pulse immediately if this country itself is the newly elected chair;
					# otherwise wait for the new chair's own monthly pulse (or next tick) to avoid double execution.
					if = {
						limit = { this = root }
						nte_global_treaty_monthly_pulse = yes
					}
				}
			}
		}
	}
