# Naval Treaty Expanded - Scripted Journal Buttons (1.15)

nte_btn_propose_revision_vote = {
	name = "nte_btn_propose_revision_vote_name"
	desc = "nte_btn_propose_revision_vote_desc"

	possible = {
		has_variable = gnl_participating_country
		OR = {
			country_rank >= rank_value:great_power
			AND = {
				exists = global_var:global_gnl_chair_country
				this = global_var:global_gnl_chair_country
			}
		}
		custom_tooltip = {
			text = nte_btn_no_active_vote_tt
			NOT = { has_global_variable = gnl_voting_active }
		}
		custom_tooltip = {
			text = nte_btn_revision_cooldown_tt
			NOT = { has_global_variable = gnl_revision_cooldown }
		}
	}

	effect = {
		set_global_variable = gnl_voting_active
		set_global_variable = { name = gnl_voting_months_left value = 3 }
		set_global_variable = { name = gnl_vote_weighted_sum value = 0 }
		set_global_variable = { name = gnl_vote_total_weight value = 0 }
		set_global_variable = { name = gnl_revision_cooldown months = 24 }
		every_country = {
			limit = { has_variable = gnl_participating_country }
			remove_variable = nte_voted_revision_flag
			trigger_event = { id = nte_global.2 days = 1 }
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { has_variable = nte_global_over_limit_flag }
			add = 10
		}
		if = {
			limit = {
				OR = {
					scaled_debt > 0.1
					has_law_or_variant = law_type:law_merchant_navy
				}
			}
			add = 10
		}
	}
}

nte_btn_propose_isolation_sanction = {
	name = "nte_btn_propose_isolation_sanction_name"
	desc = "nte_btn_propose_isolation_sanction_desc"

	possible = {
		has_variable = gnl_participating_country
		custom_tooltip = {
			text = nte_btn_no_active_iso_tt
			NOT = { has_global_variable = gnl_isolation_vote_active }
		}
		custom_tooltip = {
			text = nte_btn_need_3_members_tt
			any_country = {
				count >= 3
				has_variable = gnl_participating_country
			}
		}
		custom_tooltip = {
			text = nte_btn_need_breaching_member_tt
			any_country = {
				has_variable = gnl_participating_country
				NOT = { this = root }
				OR = {
					has_modifier = global_naval_limitation_union_penalty
					infamy >= 50
				}
			}
		}
	}

	effect = {
		ordered_country = {
			limit = {
				has_variable = gnl_participating_country
				NOT = { this = root }
				OR = {
					has_modifier = global_naval_limitation_union_penalty
					infamy >= 50
				}
			}
			order_by = nte_country_effective_tonnage
			position = 0
			set_global_variable = { name = gnl_isolation_target value = this }
		}
		set_global_variable = gnl_isolation_vote_active
		set_global_variable = { name = gnl_isolation_months_left value = 2 }
		set_global_variable = { name = gnl_iso_yes_weight value = 0 }
		set_global_variable = { name = gnl_iso_total_weight value = 0 }
		every_country = {
			limit = {
				has_variable = gnl_participating_country
				NOT = { this = global_var:gnl_isolation_target }
			}
			remove_variable = nte_iso_voted_yes
			remove_variable = nte_iso_voted_flag
			trigger_event = { id = nte_global.3 days = 1 }
		}
	}

	ai_chance = {
		value = 10
		if = {
			limit = {
				OR = {
					has_modifier = global_naval_limitation_union_penalty
					has_modifier = nte_isolated_sanction_modifier
					infamy >= 50
				}
			}
			add = -20
		}
	}
}

nte_btn_toggle_mothball_fleet = {
	name = "nte_btn_toggle_mothball_fleet_name"
	desc = "nte_btn_toggle_mothball_fleet_desc"

	possible = {
		has_variable = gnl_participating_country
		OR = {
			has_modifier = nte_mothballed_fleet_modifier
			is_at_war = no
		}
		nte_country_effective_tonnage > 0
		NOT = { has_variable = nte_mothball_cooldown }
	}

	effect = {
		set_variable = { name = nte_mothball_cooldown months = 12 }
		if = {
			limit = { has_modifier = nte_mothballed_fleet_modifier }
			remove_modifier = nte_mothballed_fleet_modifier
		}
		else = {
			add_modifier = { name = nte_mothballed_fleet_modifier }
		}
		nte_update_country_tonnage_vars = yes
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				NOT = { has_modifier = nte_mothballed_fleet_modifier }
				has_variable = nte_global_over_limit_flag
			}
			add = 80
		}
		if = {
			limit = {
				has_modifier = nte_mothballed_fleet_modifier
				NOR = {
					has_variable = nte_global_over_limit_flag
					has_variable = nte_global_near_limit_flag
				}
			}
			add = 60
		}
	}
}
