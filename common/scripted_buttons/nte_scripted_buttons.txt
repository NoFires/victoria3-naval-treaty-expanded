# Naval Treaty Expanded - Scripted Journal Buttons (1.15)

nte_btn_propose_revision_vote = {
	name = "nte_btn_propose_revision_vote_name"
	desc = "nte_btn_propose_revision_vote_desc"

	possible = {
		has_variable = gnl_participating_country
		OR = {
			country_rank >= rank_value:great_power
			AND = {
				exists = global_var:global_gnl_chair_country
				this = global_var:global_gnl_chair_country
			}
		}
		custom_tooltip = {
			text = nte_btn_no_active_vote_tt
			NOT = { has_global_variable = gnl_voting_active }
		}
		custom_tooltip = {
			text = nte_btn_revision_cooldown_tt
			NOT = { has_global_variable = gnl_revision_cooldown }
		}
	}

	effect = {
		set_global_variable = gnl_voting_active
		set_global_variable = { name = gnl_voting_months_left value = 4 }
		set_global_variable = { name = gnl_vote_weighted_sum value = 0 }
		set_global_variable = { name = gnl_vote_total_weight value = 0 }
		set_global_variable = { name = gnl_revision_cooldown months = 24 }
		every_country = {
			limit = { has_variable = gnl_participating_country }
			remove_variable = nte_voted_revision_flag
			trigger_event = { id = nte_global.2 days = 1 }
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { has_variable = nte_global_over_limit_flag }
			add = 10
		}
		if = {
			limit = {
				OR = {
					scaled_debt > 0.1
					has_law_or_variant = law_type:law_merchant_navy
				}
			}
			add = 10
		}
	}
}

nte_btn_propose_isolation_sanction = {
	name = "nte_btn_propose_isolation_sanction_name"
	desc = "nte_btn_propose_isolation_sanction_desc"

	possible = {
		has_variable = gnl_participating_country
		custom_tooltip = {
			text = nte_btn_no_active_iso_tt
			NOT = { has_global_variable = gnl_isolation_vote_active }
		}
		custom_tooltip = {
			text = nte_btn_iso_cooldown_tt
			NOT = { has_global_variable = gnl_isolation_cooldown }
		}
		custom_tooltip = {
			text = nte_btn_need_3_members_tt
			any_country = {
				count >= 3
				has_variable = gnl_participating_country
			}
		}
		custom_tooltip = {
			text = nte_btn_need_breaching_member_tt
			any_country = {
				has_variable = gnl_participating_country
				NOT = { this = root }
				NOR = {
					has_treaty_alliance_with = { TARGET = root }
					has_treaty_defensive_pact_with = { TARGET = root }
				}
				OR = {
					has_modifier = global_naval_limitation_union_penalty
					infamy >= 50
				}
			}
		}
	}

	effect = {
		ordered_country = {
			limit = {
				has_variable = gnl_participating_country
				NOT = { this = root }
				NOR = {
					has_treaty_alliance_with = { TARGET = root }
					has_treaty_defensive_pact_with = { TARGET = root }
				}
				OR = {
					has_modifier = global_naval_limitation_union_penalty
					infamy >= 50
				}
			}
			order_by = nte_country_effective_tonnage
			position = 0
			set_global_variable = { name = gnl_isolation_target value = this }
		}
		set_global_variable = gnl_isolation_vote_active
		set_global_variable = { name = gnl_isolation_months_left value = 3 }
		set_global_variable = { name = gnl_isolation_cooldown months = 24 }
		set_global_variable = { name = gnl_iso_yes_weight value = 0 }
		set_global_variable = { name = gnl_iso_total_weight value = 0 }
		every_country = {
			limit = {
				has_variable = gnl_participating_country
				NOT = { global_var:gnl_isolation_target ?= this }
			}
			remove_variable = nte_iso_voted_yes
			remove_variable = nte_iso_voted_flag
			trigger_event = { id = nte_global.3 days = 1 }
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				NOR = {
					has_modifier = global_naval_limitation_union_penalty
					has_modifier = nte_isolated_sanction_modifier
					infamy >= 50
				}
				any_country = {
					has_variable = gnl_participating_country
					NOT = { this = root }
					NOR = {
						has_treaty_alliance_with = { TARGET = root }
						has_treaty_defensive_pact_with = { TARGET = root }
					}
					OR = {
						has_modifier = global_naval_limitation_union_penalty
						infamy >= 50
					}
					OR = {
						any_rival_country = { this = root }
						"relations(root)" < 0
					}
				}
			}
			add = 15
		}
	}
}

nte_btn_invoke_escalator_clause = {
	name = "nte_btn_invoke_escalator_clause_name"
	desc = "nte_btn_invoke_escalator_clause_desc"

	possible = {
		has_variable = gnl_participating_country
		custom_tooltip = {
			text = nte_btn_escalator_threat_tt
			has_global_variable = gnl_escalator_threat_active
		}
		NOT = { has_modifier = nte_escalator_clause_modifier }
	}

	effect = {
		add_modifier = { name = nte_escalator_clause_modifier years = 3 }
		set_variable = { name = nte_escalator_clause_active years = 3 }
		if = { limit = { has_variable = overhaul_arms_race_paused_by_treaty } remove_variable = overhaul_arms_race_paused_by_treaty }
		if = { limit = { has_variable = naval_arms_race_frozen_cd } remove_variable = naval_arms_race_frozen_cd }
		nte_update_country_tonnage_vars = yes
		debug_log = "V3NAVY_EVENT|date=[TimeKeeper.GetCurrentDate.GetString]|event=ESCALATOR_INVOKED|country=[THIS.GetCountry.GetNameNoFormatting]"
	}

	# The threat is active almost permanently (any big non-member qualifies), so the AI only invokes under real pressure.
	ai_chance = {
		value = 0
		if = {
			limit = {
				OR = {
					has_variable = nte_global_near_limit_flag
					has_variable = nte_global_over_limit_flag
					is_at_war = yes
				}
			}
			add = 20
		}
	}
}

nte_btn_peaceful_withdrawal = {
	name = "nte_btn_peaceful_withdrawal_name"
	desc = "nte_btn_peaceful_withdrawal_desc"

	possible = {
		has_variable = gnl_participating_country
		OR = {
			custom_tooltip = {
				text = nte_btn_escalator_threat_tt
				has_global_variable = gnl_escalator_threat_active
			}
			influence >= 100
		}
	}

	effect = {
		if = {
			limit = { NOT = { has_global_variable = gnl_escalator_threat_active } }
			change_infamy = 5
			add_modifier = { name = nte_prestige_loss_small_modifier years = 2 }
			add_modifier = { name = nte_withdrawal_cost_modifier years = 2 }
		}
		debug_log = "V3NAVY_EVENT|date=[TimeKeeper.GetCurrentDate.GetString]|event=PEACEFUL_WITHDRAWAL|country=[THIS.GetCountry.GetNameNoFormatting]"
		set_variable = { name = nte_withdrawn_from_treaty months = 24 }
		save_scope_as = withdrawing_country
		nte_global_on_withdrawal = yes
	}

	# AI members stay at least 5 years (nte_ai_withdrawal_lock) and then leave only under sustained pressure.
	ai_chance = {
		value = 0
		if = {
			limit = {
				NOT = { has_variable = nte_ai_withdrawal_lock }
				has_global_variable = gnl_escalator_threat_active
				has_variable = nte_escalator_clause_active
				has_variable = nte_global_over_limit_flag
				OR = {
					has_law_or_variant = law_type:law_professional_navy
					has_strategy = ai_strategy_colonial_expansion
					has_strategy = ai_strategy_territorial_expansion
				}
			}
			add = 10
		}
		else_if = {
			limit = {
				NOT = { has_variable = nte_ai_withdrawal_lock }
				influence >= 250
				infamy < 20
				has_modifier = global_naval_limitation_union_penalty
				has_law_or_variant = law_type:law_professional_navy
				scaled_debt <= 0.05
				net_fixed_income > 0
			}
			add = 3
		}
	}
}

nte_btn_toggle_mothball_fleet = {
	name = "nte_btn_toggle_mothball_fleet_name"
	desc = "nte_btn_toggle_mothball_fleet_desc"

	possible = {
		has_variable = gnl_participating_country
		OR = {
			has_modifier = nte_mothballed_fleet_modifier
			is_at_war = no
		}

		NOT = { has_variable = nte_mothball_cooldown }

		set_variable = { name = nte_mothball_cooldown months = 12 }

			limit = { has_modifier = nte_mothballed_fleet_modifier }

		}
		else = {
			add_modifier = { name = nte_mothballed_fleet_modifier }
		}
		nte_update_country_tonnage_vars = yes
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				NOT = { has_modifier = nte_mothballed_fleet_modifier }
				has_variable = nte_global_over_limit_flag
			}
			add = 80
		}
		if = {
			limit = {
				has_modifier = nte_mothballed_fleet_modifier
				NOR = {
					has_variable = nte_global_over_limit_flag
					has_variable = nte_global_near_limit_flag
				}
			}
			add = 60
		}
	}
}

nte_btn_submarine_protocol = {
	name = "nte_btn_submarine_protocol_name"
	desc = "nte_btn_submarine_protocol_desc"

	possible = {
		has_variable = gnl_participating_country
		NOT = { has_modifier = nte_submarine_protocol_modifier }
		NOT = { has_modifier = nte_isolated_sanction_modifier }
		NOT = { has_variable = nte_tonnage_falsification_active }
		influence >= 25
	}

	effect = {
		add_modifier = { name = nte_submarine_protocol_modifier years = 5 }
		set_variable = { name = nte_submarine_protocol_active years = 5 }
		if = {
			limit = { NOT = { has_global_variable = gnl_treaty_stability } }
			set_global_variable = { name = gnl_treaty_stability value = 75 }
		}
		change_global_variable = { name = gnl_treaty_stability add = 3 }
		nte_sync_global_stability_bars = yes
		debug_log = "V3NAVY_EVENT|date=[TimeKeeper.GetCurrentDate.GetString]|event=SUBMARINE_PROTOCOL|country=[THIS.GetCountry.GetNameNoFormatting]"
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				is_at_war = no
				influence >= 100
			}
			add = 12
			if = {
				limit = { infamy > 15 }
				add = 25
			}
			if = {
				limit = { has_law_or_variant = law_type:law_merchant_navy }
				add = 30
			}
			if = {
				limit = { has_law_or_variant = law_type:law_diplomatic_navy }
				add = 20
			}
			if = {
				limit = { has_law_or_variant = law_type:law_jeune_ecole }
				add = -40
			}
			if = {
				limit = {
					infamy <= 25
					any_scope_ship = {
						is_ship_type = ship_type:ship_type_submarine
					}
				}
				add = -35
			}
		}
		# A submarine-heavy raider gutting its own arm is self-defeating: the protocol cuts convoy raiding and submarine damage
		if = {
			limit = {
				any_scope_ship = {
					count >= 5
					is_ship_type = ship_type:ship_type_submarine
				}
			}
			multiply = 0
		}
	}
}
