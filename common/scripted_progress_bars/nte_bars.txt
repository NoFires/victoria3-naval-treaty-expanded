# Naval Treaty Expanded - 1.15 Scripted Progress Bars

nte_treaty_stability_bar = {
	name = "nte_treaty_stability_bar_name"
	desc = "nte_treaty_stability_bar_desc"
	second_desc = "nte_treaty_stability_bar_second_desc"

	progress_texture = "gfx/interface/progressbar/progressbar_green.dds"
	background_texture = "gfx/interface/progressbar/progressbar_bad.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_green_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_green_glow_line.dds"
	marker = yes

	yearly_progress = {
		scope:journal_entry ?= {
			progress_condition = {
				limit = {
					exists = global_var:gnl_stability_drift
					NOT = { global_var:gnl_stability_drift = 0 }
				}
				add = {
					desc = "nte_stability_drift_tt"
					value = global_var:gnl_stability_drift
				}
			}
		}
	}

	start_value = 75
	min_value = 0
	max_value = 100
}
