nte_bilateral.2 = {
	type = country_event
	placement = root
	title = nte_bilateral.2.t
	desc = nte_bilateral.2.d
	flavor = nte_bilateral.2.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 2

	trigger = {
		exists = scope:nte_enforcer
		scope:nte_enforcer = { is_country_alive = yes }
	}

	option = {
		name = nte_bilateral.2.a # Back down and accept naval slowdown
		if = {
			limit = { has_variable = nte_bilateral_restricted_flag }
			set_variable = { name = nte_bilateral_grace_months value = 0 }
			set_variable = { name = nte_bilateral_crisis_triggered months = 36 }
		}
		if = {
			limit = { has_variable = nte_bilateral_formally_broken_flag }
			remove_variable = nte_bilateral_formally_broken_flag
			add_modifier = { name = nte_bilateral_capital_slowdown years = 3 }
		}
		else_if = {
			limit = { NOT = { has_modifier = nte_bilateral_breach_penalty } }
			add_modifier = { name = nte_bilateral_capital_slowdown years = 3 }
		}
		add_modifier = { name = nte_prestige_loss_small_modifier years = 2 }
		ai_chance = {
			base = 30
			modifier = {
				trigger = { navy_size < scope:nte_enforcer.navy_size }
				add = 40
			}
		}
	}

	option = {
		name = nte_bilateral.2.b # Defy the ultimatum!
		default_option = yes
		change_infamy = 12
		change_relations = { country = scope:nte_enforcer value = -50 }
		scope:nte_enforcer = {
			trigger_event = { id = nte_bilateral.3 days = 5 }
		}
		ai_chance = { base = 40 }
	}
}

# Stage 3 (Low Relations): Enforcer's Final Choice - Sanctions or Armed Intervention
nte_bilateral.3 = {
	type = country_event
	placement = root
	title = nte_bilateral.3.t
	desc = nte_bilateral.3.d
	flavor = nte_bilateral.3.f

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 2

	trigger = {
		exists = scope:nte_breaker
		scope:nte_breaker = { is_country_alive = yes }
	}

	option = {
		name = nte_bilateral.3.a # Impose Trade & Diplomatic Sanctions
		default_option = yes
		scope:nte_breaker = {
			if = {
				limit = { has_modifier = nte_bilateral_capital_slowdown }
				remove_modifier = nte_bilateral_capital_slowdown
			}
			add_modifier = { name = nte_bilateral_breach_penalty years = 5 }
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = nte_bilateral.3.b # Launch Punitive Naval Diplomatic Play!
		trigger = {
			exists = scope:nte_breaker
			is_at_war = no
			scope:nte_breaker = {
				is_country_alive = yes
				is_at_war = no
				NOT = { is_subject_of = root }
			}
			NOT = { is_subject_of = scope:nte_breaker }
			NOT = { has_truce_with = scope:nte_breaker }
			NOT = { has_treaty_alliance_with = { TARGET = scope:nte_breaker } }
			NOT = { has_treaty_defensive_pact_with = { TARGET = scope:nte_breaker } }
		}
		create_diplomatic_play = {
			name = nte_punitive_naval_play
			target_country = scope:nte_breaker
			type = dp_humiliation
		}
		ai_chance = {
			base = 25
			modifier = {
				trigger = { navy_size > scope:nte_breaker.navy_size }
				add = 30
			}
		}
	}

	option = {
		name = nte_bilateral.3.c # Stand down
		add_modifier = { name = nte_bilateral_appeasement_prestige_loss years = 3 }
		ai_chance = { base = 10 }
	}
}

					divide = 100

			}
		}

# 21: Bilateral Naval Treaty 10-Year Expiration & Renegotiation
nte_bilateral.21 = {
	type = country_event
	placement = root
	title = nte_bilateral.21.t
	desc = nte_bilateral.21.d
	flavor = nte_bilateral.21.f

	event_image = {
		video = "africa_diplomats_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 2

	trigger = {
		has_variable = nte_bilateral_renewal_target
		exists = var:nte_bilateral_renewal_target
		var:nte_bilateral_renewal_target = {
			is_country_alive = yes
			has_variable = nte_bilateral_restricted_flag
		}
	}

	option = {
		name = nte_bilateral.21.a # Renew the bilateral limitation for another decade
		default_option = yes
		if = {
			limit = { exists = var:nte_bilateral_renewal_target }
			change_relations = { country = var:nte_bilateral_renewal_target value = 10 }
			add_modifier = { name = nte_bilateral_pressure_cost_modifier months = 12 }
			nte_bilateral_on_renewal = yes
		}
		remove_variable = nte_bilateral_renewal_target
		ai_chance = {
			base = 45
		}
	}

	option = {
		name = nte_bilateral.21.b # Ease the ratio to 75% as a goodwill gesture
		trigger = { exists = var:nte_bilateral_renewal_target }
		change_relations = { country = var:nte_bilateral_renewal_target value = 20 }
		var:nte_bilateral_renewal_target = {
			set_variable = { name = nte_bilateral_ratio_pct value = 75 }
			set_variable = {
				name = nte_bilateral_cap
				value = {
					value = root.nte_country_effective_tonnage
					multiply = 75
					divide = 100
					min = 20
					round = yes
				}
			}
		}
		nte_bilateral_on_renewal = yes
		remove_variable = nte_bilateral_renewal_target
		ai_chance = {
			base = 30
			modifier = {
				trigger = {
					exists = var:nte_bilateral_renewal_target
					"relations(scope:nte_bilateral_target)" >= 10
				}
				add = 25
			}
		}
	}

	option = {
		name = nte_bilateral.21.c # Allow the bilateral treaty to lapse peacefully
		add_modifier = { name = nte_prestige_gain_modifier years = 2 }
		if = {
			limit = { exists = var:nte_bilateral_renewal_target }
			change_relations = { country = var:nte_bilateral_renewal_target value = 25 }
		}
		if = { limit = { has_modifier = nte_bilateral_pressure_cost_modifier } remove_modifier = nte_bilateral_pressure_cost_modifier }
		# 1.15.1 has no scripted treaty-termination effect: fail requirement_to_maintain instead,
		# so the engine breaks the article on its next update and on_break runs the cleanup.
		if = {
			limit = { exists = var:nte_bilateral_renewal_target }
			var:nte_bilateral_renewal_target = { set_variable = nte_bilateral_lapsed_flag }
		}
		remove_variable = nte_bilateral_renewal_target
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					exists = var:nte_bilateral_renewal_target
					"relations(scope:nte_bilateral_target)" >= 30
				}
				add = 30
			}
		}
	}
}
